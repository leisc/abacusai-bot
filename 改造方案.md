1. **创建本地 Vault**
```typescript
// packages/agent/src/vault/local-vault.ts
import * as fs from 'fs/promises';
import * as crypto from 'crypto';
import * as path from 'path';

export class LocalSecureVault {
  private vaultPath: string;
  private encryptionKey: string;
  
  constructor() {
    this.vaultPath = path.join(
      LocalDataPolicy.getDataDir(),
      'vault'
    );
    this.encryptionKey = process.env.ABACUSAI_BOT_VAULT_KEY || 
                         this.loadOrGenerateKey();
  }
  
  async setConnectorToken(
    connector: string,
    service: string,
    token: string
  ): Promise<void> {
    const encrypted = this.encrypt(token);
    const filepath = path.join(
      this.vaultPath,
      `${connector}-${service}.enc`
    );
    await fs.writeFile(filepath, encrypted);
  }
  
  async getConnectorToken(
    connector: string,
    service: string
  ): Promise<string | null> {
    const filepath = path.join(
      this.vaultPath,
      `${connector}-${service}.enc`
    );
    try {
      const encrypted = await fs.readFile(filepath, 'utf-8');
      return this.decrypt(encrypted);
    } catch {
      return null;
    }
  }
  
  private encrypt(data: string): string {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv(
      'aes-256-gcm',
      Buffer.from(this.encryptionKey),
      iv
    );
    const encrypted = cipher.update(data, 'utf-8', 'hex') +
                      cipher.final('hex');
    const tag = cipher.getAuthTag();
    return `${iv.toString('hex')}:${encrypted}:${tag.toString('hex')}`;
  }
  
  private decrypt(data: string): string {
    const [ivHex, encrypted, tagHex] = data.split(':');
    const iv = Buffer.from(ivHex, 'hex');
    const tag = Buffer.from(tagHex, 'hex');
    const decipher = crypto.createDecipheriv(
      'aes-256-gcm',
      Buffer.from(this.encryptionKey),
      iv
    );
    decipher.setAuthTag(tag);
    return decipher.update(encrypted, 'hex', 'utf-8') +
           decipher.final('utf-8');
  }
  
  private loadOrGenerateKey(): string {
    const keyFile = path.join(
      LocalDataPolicy.getDataDir(),
      '.vault-key'
    );
    try {
      return fs.readFileSync(keyFile, 'utf-8').trim();
    } catch {
      const key = crypto.randomBytes(32).toString('hex');
      fs.writeFileSync(keyFile, key, { mode: 0o600 });
      return key;
    }
  }
}
```

### **第五层：移除更新机制中的 Abacus 依赖**

#### 当前问题
- `apps/updater/` 使用 TUF 框架，与 Abacus.AI 更新服务器通信
- 用户无法审计更新内容

#### 改造方案

```typescript
// apps/updater/standalone-update.ts
export class StandaloneUpdater {
  private releaseUrl = 'https://github.com/abacusai/abacusai-bot/releases';
  
  async checkUpdates(): Promise<Release | null> {
    // 直接从 GitHub Releases 获取（透明、可审计）
    const response = await fetch(
      `${this.releaseUrl}/latest`,
      { headers: { Accept: 'application/json' } }
    );
    const release = await response.json();
    
    // 校验签名（GPG/RSA）
    const verified = await this.verifySignature(
      release.body,
      release.signatures[0]
    );
    
    if (!verified) {
      throw new Error('Release signature verification failed');
    }
    
    return release;
  }
  
  private async verifySignature(
    body: string,
    signature: string
  ): Promise<boolean> {
    // 使用公开的 GPG 密钥验证
    const publicKey = await fetch(
      'https://github.com/abacusai.gpg'
    ).then(r => r.text());
    
    // 使用 openpgp.js 验证
    const verified = await OpenPGP.verify({
      message: body,
      publicKeys: [publicKey],
      signature
    });
    
    return verified.isValid;
  }
}
```
